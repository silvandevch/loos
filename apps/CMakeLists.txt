if(NOT WIN32)
    message(FATAL_ERROR "LoOS desktop apps are PE targets; use a Windows or MinGW-w64 toolchain")
endif()

enable_language(RC)

add_executable(locmd loos-cmd/cmd.c)
set_target_properties(locmd PROPERTIES OUTPUT_NAME "LoCmd")
target_compile_definitions(locmd PRIVATE UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS)
if(MSVC)
    target_compile_options(locmd PRIVATE /W0)
else()
    target_compile_options(locmd PRIVATE -w)
    target_link_options(locmd PRIVATE -municode)
endif()
target_link_libraries(locmd PRIVATE kernel32 user32 shell32 advapi32 iphlpapi netapi32)

add_executable(lonotepad WIN32 loos-notepad/notepad.c loos-notepad/notepad.rc)
set_target_properties(lonotepad PROPERTIES OUTPUT_NAME "LoNotepad")
target_compile_definitions(lonotepad PRIVATE UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS)
target_include_directories(lonotepad PRIVATE loos-notepad)
if(MSVC)
    target_compile_options(lonotepad PRIVATE /W4)
else()
    target_compile_options(lonotepad PRIVATE -Wall -Wextra)
    target_link_options(lonotepad PRIVATE -municode)
endif()
target_link_libraries(lonotepad PRIVATE comdlg32 shell32 comctl32 winspool gdi32 ole32 uxtheme user32 kernel32 advapi32)

add_executable(lotaskmgr WIN32 loos-taskmgr/taskmgr.c loos-taskmgr/taskmgr.rc)
set_target_properties(lotaskmgr PROPERTIES OUTPUT_NAME "LoTaskMgr")
target_compile_definitions(lotaskmgr PRIVATE UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS)
target_include_directories(lotaskmgr PRIVATE loos-taskmgr)
if(MSVC)
    target_compile_options(lotaskmgr PRIVATE /W4)
else()
    target_compile_options(lotaskmgr PRIVATE -Wall -Wextra)
    target_link_options(lotaskmgr PRIVATE -municode)
endif()
target_link_libraries(lotaskmgr PRIVATE comctl32 comdlg32 user32 kernel32 gdi32 shell32 advapi32 psapi wtsapi32 version shlwapi iphlpapi pdh setupapi)
