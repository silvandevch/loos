if(BUILD_TESTING)
    function(loos_pkg_test name)
        add_executable(${name} ${name}.c)
        target_link_libraries(${name} PRIVATE loos_pkg)
        target_compile_options(${name} PRIVATE ${LOOS_C_WARNINGS})
        add_test(NAME pkg.${name} COMMAND ${name})
    endfunction()
    loos_pkg_test(deb_test)
    loos_pkg_test(dpkgdb_test)
    loos_pkg_test(solver_test)
    loos_pkg_test(apt_test)
    loos_pkg_test(pgp_test)
endif()
