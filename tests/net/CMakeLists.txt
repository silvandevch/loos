if(BUILD_TESTING)
    function(loos_net_test name)
        add_executable(${name} ${name}.c)
        target_link_libraries(${name} PRIVATE loos_net)
        target_compile_options(${name} PRIVATE ${LOOS_C_WARNINGS})
        add_test(NAME net.${name} COMMAND ${name})
    endfunction()
    loos_net_test(netutil_test)
    loos_net_test(dns_test)
    loos_net_test(dhcp_test)
    loos_net_test(fw_test)
    loos_net_test(http_test)
endif()
