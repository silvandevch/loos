if(BUILD_TESTING)
    function(loos_shell_test name)
        add_executable(${name} ${name}.c)
        target_link_libraries(${name} PRIVATE loos_shell)
        target_compile_options(${name} PRIVATE ${LOOS_C_WARNINGS})
        add_test(NAME shell.${name} COMMAND ${name})
    endfunction()
    loos_shell_test(compositor_test)

    loos_shell_test(a11y_test)
    loos_shell_test(session_test)

    loos_shell_test(installer_test)
