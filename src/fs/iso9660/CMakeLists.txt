if(TARGET loos_rust)
    # Hosted builds: ISO 9660 / Joliet / Rock Ridge via crates/loos-fs behind the same C API.
    add_library(loos_iso9660 INTERFACE)
    target_include_directories(loos_iso9660 INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
    target_compile_definitions(loos_iso9660 INTERFACE LOOS_RUST_ISO9660=1)
    target_link_libraries(loos_iso9660 INTERFACE loos_rust)
else()
    add_library(loos_iso9660 STATIC iso9660.c)
    target_include_directories(loos_iso9660 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
    target_compile_options(loos_iso9660 PRIVATE ${LOOS_C_WARNINGS})

    if(TARGET loos_kcrt)
        target_link_libraries(loos_iso9660 PUBLIC loos_kcrt)
    endif()
endif()
